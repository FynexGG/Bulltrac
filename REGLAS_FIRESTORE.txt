rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Solo se permite el documento de datos de Control Vehicular; todo lo demás queda bloqueado.
    match /control_vehicular/datos {
      allow read: if true;
      allow write: if request.resource.data.keys().hasOnly(['datos', 'actualizado', 'cliente'])
        && request.resource.data.datos is string
        && request.resource.data.datos.size() < 900000;
    }
  }
}
